        const R2_BASE = "https://pub-b24d092a7bf34fd9bf22befba89de35f.r2.dev";
        const SUPABASE_URL = "https://lxdmfaxxxfyzbpzvniyi.supabase.co";
        const SUPABASE_KEY = "sb_publishable_iPykeIMk376fBXL0b-hGjA_FkvaDcVp";

        const MAPA_VOZES = {
            en: ["v1", "v2", "v3", "v4", "v5", "v6"],

            fr: ["v1", "v2", "v3", "v4"],
            it: ["v1", "v2", "v3", "v4"],
            ge: ["v1", "v2", "v3", "v4"],
            pt: ["v1", "v2", "v3", "v4"],
            zh: ["v1", "v2", "v3", "v4"]
        };

        let frases = [];
        let indiceAtual = 0;
        let vozAtual = "v1";
        let audio = new Audio();
        let timerAutoplay = null;

        async function carregarFrases() {
            const badge = document.getElementById("badgeProgresso");
            const textoPrincipal = document.getElementById("textoPrincipal");
            badge.innerText = "Baixando frases...";

            try {
                const headers = {
                    "apikey": SUPABASE_KEY,
                    "Authorization": `Bearer ${SUPABASE_KEY}`
                };

                // Busca em dois lotes para cobrir todas as 1.200 frases sem travar no limite de 1.000 do Supabase
                const [resp1, resp2] = await Promise.all([
                    fetch(`${SUPABASE_URL}/rest/v1/sentences?select=*&order=id.asc&limit=1000&offset=0`, { headers }),
                    fetch(`${SUPABASE_URL}/rest/v1/sentences?select=*&order=id.asc&limit=1000&offset=1000`, { headers })
                ]);

                if (!resp1.ok) throw new Error(`Falha no lote 1: HTTP ${resp1.status}`);

                const data1 = await resp1.json();
                const data2 = resp2.ok ? await resp2.json() : [];
                frases = [...data1, ...data2];

                if (!frases.length) throw new Error("O banco de dados não retornou frases.");

                filtrarFrases();
            } catch (err) {
                console.error("[Auditor] Erro de conexão:", err);
                badge.innerText = "Erro de conexão";
                badge.style.backgroundColor = "var(--danger)";
                badge.style.color = "#ffffff";
                textoPrincipal.innerHTML = `<span style="color: var(--danger); font-size: 1.1rem;">Falha ao carregar frases: ${err.message}</span>`;
            }
        }

        function filtrarFrases() {
            const nivel = document.getElementById("selectNivel").value;
            const idioma = document.getElementById("selectIdioma").value;

            window.frasesFiltradas = frases.filter(f => {
                const matchNivel = (nivel === "TODOS" || f.level === nivel);
                const temIdioma = Boolean(f[idioma]);
                return matchNivel && temIdioma;
            });

            indiceAtual = 0;
            atualizarVozesDisponiveis();
            atualizarCard();
        }

        function atualizarVozesDisponiveis() {
            const idioma = document.getElementById("selectIdioma").value;
            const vozes = MAPA_VOZES[idioma] || ["v1"];
            if (!vozes.includes(vozAtual)) vozAtual = "v1";

            const grade = document.getElementById("gradeVozes");
            grade.innerHTML = "";
            vozes.forEach(v => {
                const btn = document.createElement("button");
                btn.className = "btn-voz" + (v === vozAtual ? " ativo" : "");
                btn.innerText = v.toUpperCase();
                btn.onclick = () => {
                    vozAtual = v;
                    atualizarVozesDisponiveis();
                    tocarAudio();
                };
                grade.appendChild(btn);
            });
        }

        function atualizarCard() {
            const lista = window.frasesFiltradas || [];
            if (lista.length === 0) {
                document.getElementById("textoPrincipal").innerText = "Nenhuma frase encontrada para este filtro.";
                document.getElementById("textoTraducao").innerText = "";
                return;
            }

            const item = lista[indiceAtual];
            const idioma = document.getElementById("selectIdioma").value;

            document.getElementById("textoPrincipal").innerText = item[idioma] || item.texto || "";
            document.getElementById("textoTraducao").innerText = item.pt || item.topic_pt || "";
            document.getElementById("labelTopico").innerText = `ID #${item.id} · ${item.level} · ${item.topic_pt || item.topic_en || ''}`;
            document.getElementById("labelContador").innerText = `${indiceAtual + 1} / ${lista.length}`;

            const badge = document.getElementById("badgeProgresso");
            badge.innerText = `${lista.length} frases prontas`;
            badge.style.backgroundColor = "#334155";
            badge.style.color = "var(--muted)";
        }

        function tocarAudio() {
            clearTimeout(timerAutoplay);
            const lista = window.frasesFiltradas || [];
            if (lista.length === 0) return;

            const item = lista[indiceAtual];
            const idioma = document.getElementById("selectIdioma").value;
            const pasta = idioma === "pi" ? "zh" : idioma;
            const url = `${R2_BASE}/${pasta}/${item.id}_${vozAtual}.mp3?v=${Date.now()}`;

            audio.pause();
            audio = new Audio(url);
            audio.playbackRate = parseFloat(document.getElementById("selectVelocidade").value);

            document.getElementById("btnPlay").innerText = "⏸ Pausar";

            audio.onended = () => {
                document.getElementById("btnPlay").innerText = "▶ Reproduzir";
                if (document.getElementById("checkAutoplay").checked) {
                    timerAutoplay = setTimeout(() => {
                        avancar();
                        tocarAudio();
                    }, 1000);
                }
            };

            audio.onerror = () => {
                document.getElementById("btnPlay").innerText = "Erro Áudio";
                if (document.getElementById("checkAutoplay").checked) {
                    timerAutoplay = setTimeout(() => {
                        avancar();
                        tocarAudio();
                    }, 1500);
                }
            };

            audio.play().catch(() => {
                document.getElementById("btnPlay").innerText = "▶ Reproduzir";
            });
        }

        function pausarOuTocar() {
            if (!audio.paused && audio.duration > 0) {
                audio.pause();
                document.getElementById("btnPlay").innerText = "▶ Reproduzir";
            } else {
                tocarAudio();
            }
        }

        function avancar() {
            const lista = window.frasesFiltradas || [];
            if (indiceAtual < lista.length - 1) {
                indiceAtual++;
                atualizarCard();
            }
        }

        function voltar() {
            if (indiceAtual > 0) {
                indiceAtual--;
                atualizarCard();
            }
        }

        async function sinalizarAudio() {
            const item = window.frasesFiltradas[indiceAtual];
            const idioma = document.getElementById("selectIdioma").value;
            const btn = document.getElementById("btnReportar");
            const inputProblema = document.getElementById("inputProblema");
            const textoProblema = inputProblema ? inputProblema.value.trim() : "";

            btn.innerText = "A registrar...";

            try {
                const resp = await fetch(`${SUPABASE_URL}/rest/v1/reports_frases`, {
                    method: "POST",
                    headers: {
                        "apikey": SUPABASE_KEY,
                        "Authorization": `Bearer ${SUPABASE_KEY}`,
                        "Content-Type": "application/json",
                        "Prefer": "return=minimal"
                    },
                    body: JSON.stringify([{
                        frase_id: item.id,
                        idioma_estudo: idioma,
                        idioma_origem: "pt",
                        voz: vozAtual,
                        tipo_problema: textoProblema ? `Auditoria: ${textoProblema}` : "Auditoria Manual: Áudio com defeito",
                        observacao: `Sinalizado via Auditor Web (${item.id}_${vozAtual}.mp3)${textoProblema ? ` - ${textoProblema}` : ""}`,

def sanitizar_palavra_audio(palavra):
    p = (palavra or "").strip().lower().replace("ß", "ss")
    sem_pontuacao = re.sub(r'[.,!?;:¿¡"“”`{}()[\]\-—…，。！？；：、«»/\\~*]', '', p)
    sem_pontuacao = re.sub(r"['’]", "_", sem_pontuacao)
    if not sem_pontuacao:
        return ""

    nfd = unicodedata.normalize("NFD", sem_pontuacao)
    mapa_diacriticos = {
        "\u0300": "grave", "\u0301": "acute", "\u0302": "circ", "\u0303": "tilde",
        "\u0308": "uml", "\u0327": "ced", "\u0304": "macron", "\u030c": "caron"
    }

    base_chars, marcas = [], []
    ultimo_char_base = ""
    for char in nfd:
        if char in mapa_diacriticos:
            marcas.append(f"{ultimo_char_base}_{mapa_diacriticos[char]}")
        else:
            if not ("\u0300" <= char <= "\u036f"):
                base_chars.append(char)
                ultimo_char_base = char

    slug_base = re.sub(r"[^a-zA-Z0-9_-]", "", "".join(base_chars))
    if marcas:
        return f"{slug_base}_{'_'.join(marcas)}"
    return slug_base

def cmd_frases_f2(idioma, inicio, fim, vozes_custom=None):

    inicio = int(inicio)

    if inicio < 1 or fim > 1200 or inicio > fim:
        print("[ERRO] Faixa de IDs inválida. Deve estar entre 1 e 1200.")

    todas_vozes = VOZES_FRASES[idioma]
    if vozes_custom:
        tags_selecionadas = [v.strip().lower() for v in vozes_custom.split(",")]
        vozes_alvo = [v for v in todas_vozes if v[0] in tags_selecionadas]
    else:
        vozes_alvo = todas_vozes

    if not vozes_alvo:
        print(f"[ERRO] Nenhuma voz válida selecionada. Opções: {[v[0] for v in todas_vozes]}")
        sys.exit(1)

    print("=" * 70)
    print(f" DÌNGLÌ - FLUXO F2: SÍNTESE E UPLOAD DE FRASES COMPLETAS ({idioma.upper()})")
    print(f" Faixa: ID {inicio} a {fim} ({fim - inicio + 1} frases)")
    print(f" Vozes: {', '.join(v[0] for v in vozes_alvo)}")
    print("=" * 70)

    frases_map = {}
    with open(CSV_SENTENCES, "r", encoding="utf-8-sig", errors="ignore") as f:
        for r in csv.DictReader(f):
            fid = r.get("id")
            if fid and fid.isdigit():
                fid_int = int(fid)
                if inicio <= fid_int <= fim:
                    txt = (r.get(idioma) or "").strip()
                    if txt:
                        frases_map[fid_int] = txt

    pasta_frases = os.path.join("audios_dingli", idioma)
    os.makedirs(pasta_frases, exist_ok=True)

    print("Iniciando síntese Edge-TTS...")
    semaforo = asyncio.Semaphore(4)
    progresso = {"ok": 0, "erros": 0}

    async def sintetizar(fid, texto, tag_voz, voz):
        final_mp3 = os.path.join(pasta_frases, f"{fid}_{tag_voz}.mp3")
        async with semaforo:
            sucesso = False
            for _ in range(3):
                try:
                    com = edge_tts.Communicate(texto, voz)
                    await com.save(final_mp3)
                    if os.path.exists(final_mp3) and os.path.getsize(final_mp3) > 0:
                        sucesso = True
                        break
                except Exception:
                    await asyncio.sleep(1)
            if sucesso:
                progresso["ok"] += 1
            else:
                progresso["erros"] += 1

    async def rodar():
        tarefas = []
        for fid, texto in frases_map.items():
            for tag_voz, voz in vozes_alvo:
                tarefas.append(sintetizar(fid, texto, tag_voz, voz))
        await asyncio.gather(*tarefas)

    asyncio.run(rodar())
    print(f"✔ Síntese concluída: {progresso['ok']} arquivos gerados | Falhas: {progresso['erros']}")

    print("[PRÓXIMO PASSO]")
    print("Registre a versão de áudio no Supabase (sem impacto no progresso dos alunos):")

function validarTextoSemJargao(texto) {
  const textoMin = texto.toLowerCase();
  for (const termo of PALAVRAS_PROIBIDAS) {
    if (textoMin.includes(termo.toLowerCase())) {
      return { valido: false, termo };
    }
  }
  return { valido: true };
}

async function chamarGemini(blocoFrases, tentativa = 1) {
  const prompt = `
Você é um professor experiente e muito didático.
Explique as frases em Francês para alunos brasileiros como um amigo experiente que mora fora explicando a lógica prática do dia a dia, de forma clara, natural e sem termos técnicos de gramática (PROIBIDO usar: ${PALAVRAS_PROIBIDAS.join(', ')}).

Regras obrigatórias:
1. Explicação: Curta, direta (2 a 4 linhas), explicando o raciocínio real de como o francês constrói a ideia.
2. Breakdown: Decomposição EXAUSTIVA de 100% das palavras e partículas da frase. Nenhuma palavra ou partícula pode ficar de fora.
3. Formato: Retorne ESTRITAMENTE um objeto JSON onde cada chave é o ID da frase (como string) contendo "explanation" e "breakdown" (array com objetos contendo "word" e "meaning").

${EXEMPLOS_PADRAO_OURO}

Gere agora para este lote:
${JSON.stringify(blocoFrases, null, 2)}
`;

  try {
    const resposta = await fetch(GEMINI_URL, {
      method: "POST",
      headers: {
        "Content-Type": "application/json",
        "x-goog-api-key": API_KEY
      },
      body: JSON.stringify({
        contents: [{ parts: [{ text: prompt }] }],
        generationConfig: { responseMimeType: "application/json", temperature: 0.2 }
      })
    });

    if (!resposta.ok) {
      throw new Error(`HTTP ${resposta.status}: ${await resposta.text()}`);
    }

    const data = await resposta.json();
    const textoGerado = data.candidates?.[0]?.content?.parts?.[0]?.text;
    const jsonParsed = JSON.parse(textoGerado);

    // Auditoria para barrar jargão
    const auditoria = validarTextoSemJargao(textoGerado);
    if (!auditoria.valido) {
      if (tentativa <= 3) {
        console.warn(`[Auditoria] Jargão encontrado ('${auditoria.termo}'). Repetindo lote (tentativa ${tentativa + 1})...`);
        await aguardar(2000);
        return await chamarGemini(blocoFrases, tentativa + 1);
      }
    }

    return jsonParsed;
  } catch (err) {
    if (tentativa <= 4) {
      const is429 = err.message.includes('429');
      const tempoEspera = is429 ? 35000 : 4000;
      console.warn(`[API] ${is429 ? 'Limite de requisições por minuto atingido (429).' : 'Erro na chamada:'} Aguardando ${tempoEspera / 1000}s para retentar (tentativa ${tentativa + 1})...`);
      await aguardar(tempoEspera);
      return await chamarGemini(blocoFrases, tentativa + 1);
    }
    throw err;
  }
}

      const conteudo = JSON.parse(fs.readFileSync(caminhoArquivo, 'utf-8'));
      if (Object.keys(conteudo).length >= frasesTopico.length) {
        console.log(`[Checkpoint] Pulando '${nomeArquivo}' (já completo com ${frasesTopico.length} frases).`);
        return;
      }
    } catch (e) {}
  }

  console.log(`\n======================================================`);
  console.log(`Iniciando Tópico: [${nivel}] ${nomeTopico} (${frasesTopico.length} frases)`);
  console.log(`======================================================`);

  const resultadoTopico = {};
  const TAMANHO_LOTE = 5;

  for (let i = 0; i < frasesTopico.length; i += TAMANHO_LOTE) {
    const lote = frasesTopico.slice(i, i + TAMANHO_LOTE).map(f => ({
      id: f.id,
      fr: f.fr,
      pt: f.pt
    }));

    const loteNum = Math.floor(i / TAMANHO_LOTE) + 1;
    const totalLotes = Math.ceil(frasesTopico.length / TAMANHO_LOTE);
    console.log(`  -> Processando lote ${loteNum}/${totalLotes} (frases ${i + 1} a ${Math.min(i + TAMANHO_LOTE, frasesTopico.length)})...`);

    const resultadoLote = await chamarGemini(lote);
    Object.assign(resultadoTopico, resultadoLote);

    // Pausa preventiva de 4.5 segundos entre lotes para respeitar a cota de 15 RPM
    await aguardar(4500);
  }

  fs.writeFileSync(caminhoArquivo, JSON.stringify(resultadoTopico, null, 2), 'utf-8');
  console.log(`✔ Arquivo salvo com sucesso: ${nomeArquivo}`);
}

async function iniciarEsteira() {
  const pastaDestino = path.resolve(__dirname, '../public/explicacoes/fr_pt');
  if (!fs.existsSync(pastaDestino)) {
    fs.mkdirSync(pastaDestino, { recursive: true });
  }

  const NIVEIS = ['A1', 'A2', 'B1', 'B2'];

  for (const nivel of NIVEIS) {
    console.log(`\n>>> Carregando frases do nível ${nivel} do Supabase...`);
    
    // Consulta por nível (300 frases cada, sem atingir a trava de 1.000 registros do Supabase)
    const { data, error } = await supabase
      .from('sentences')
      .select('id, fr, pt, level, topic_pt')
      .eq('level', nivel)
      .not('fr', 'is', null)
      .order('id', { ascending: true });

    if (error) {
      console.error(`Erro ao buscar dados do nível ${nivel}:`, error);
      continue;
    }

    // Agrupa as 300 frases em seus 10 tópicos mantendo a ordem correta dos IDs
    const topicosDoNivel = {};
    for (const f of data) {
      const tp = f.topic_pt;
      if (!tp) continue;
      if (!topicosDoNivel[tp]) topicosDoNivel[tp] = [];
      topicosDoNivel[tp].push(f);
    }

    for (const [nomeTopico, frases] of Object.entries(topicosDoNivel)) {
      await processarTopico(nivel, nomeTopico, frases, pastaDestino);
    }
  }

  console.log('\n======================================================');
  console.log('ESTEIRA CONCLUÍDA: Todos os tópicos foram processados!');
  console.log('======================================================');
}

iniciarEsteira();
